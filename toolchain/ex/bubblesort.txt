; bubble sort
define SIZE #8
mov r1 :arr
xor r2 r2
while
	ifbreak r2 >= SIZE

	xor r3 r3
	while
		mov r4 SIZE
		sub r4 r2
		add r4 #-1
		ifbreak r3 >= r4

		; print ^
		add r3 #8

		mov out r3
		dindex
		mov out #'^'
		dwrite

		inc r3
		mov out r3
		dindex
		mov out #'^'
		dwrite

		add r3 #-9
		; -----

		mov r4 r1
		add r4 r3
		mov r5 r4
		inc r5
		mov r6 [r4]
		mov r7 [r5]
		if r6 > r7
			; print *
			add r3 #8

			mov out r3
			dindex
			mov out #'*'
			dwrite

			inc r3
			mov out r3
			dindex
			mov out #'*'
			dwrite

			add r3 #-9
			; -----

			mov [r4] r7
			mov [r5] r6

			; print num
			mov out r3
			dindex
			mov out [r4]
			dwrite

			inc r3
			mov out r3
			dindex
			mov out [r5]
			dwrite
			add r3 #-1
			; -----
		}

		; delete ^
		add r3 #8

		mov out r3
		dindex
		mov out #'\0'
		dwrite

		inc r3
		mov out r3
		dindex
		mov out #'\0'
		dwrite

		; add r3 #-9
		add r3 #-8
		; -----

		; inc r3
	}

	inc r2
}
